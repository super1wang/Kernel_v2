cmake_minimum_required(VERSION 3.25)
project(OCKInstalledB2Consumer LANGUAGES CXX)
find_package(OCK CONFIG REQUIRED COMPONENTS Data Dynamic ControlProtocol Control Runtime)
if(NOT OCK_IMPLEMENTATION_STAGE STREQUAL "B6Subset")
  message(FATAL_ERROR "B2 SDK component metadata missing")
endif()
get_target_property(native_links OCK::Runtime INTERFACE_LINK_LIBRARIES)
if(NOT native_links STREQUAL "OCK::CoreContracts;$<LINK_ONLY:bcrypt>")
  message(FATAL_ERROR "Native link closure expanded")
endif()
if(TARGET OCKThirdParty::jsoncons)
  message(FATAL_ERROR "Private backend exported")
endif()
add_executable(installed_b2 main.cpp)
target_link_libraries(installed_b2 PRIVATE OCK::Control)
