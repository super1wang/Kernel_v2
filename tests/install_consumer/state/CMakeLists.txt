cmake_minimum_required(VERSION 3.25)
project(OCKStateInstalledConsumer LANGUAGES CXX)
if(REQUIRE_COMPONENT)
  find_package(OCK 0.1.0 CONFIG REQUIRED COMPONENTS State Runtime ${REQUIRE_COMPONENT})
else()
  find_package(OCK 0.1.0 CONFIG REQUIRED COMPONENTS State Runtime)
endif()
if(NOT OCK_SDK_VERSION STREQUAL "0.1.0-dev.7" OR
   NOT OCK_IMPLEMENTATION_STAGE STREQUAL "B6Subset" OR
   NOT OCK_INSTALLATION_PROFILE MATCHES "^(StateNative|B6Subset)$")
  message(FATAL_ERROR "B6 State installed SDK metadata mismatch")
endif()
get_target_property(state_kind OCK::State TYPE)
get_target_property(state_links OCK::State INTERFACE_LINK_LIBRARIES)
if(NOT state_kind STREQUAL "STATIC_LIBRARY" OR NOT state_links STREQUAL "OCK::CoreContracts")
  message(FATAL_ERROR "State must export the CoreContracts-only public closure")
endif()
add_executable(installed_state main.cpp)
target_link_libraries(installed_state PRIVATE OCK::State OCK::Runtime)
