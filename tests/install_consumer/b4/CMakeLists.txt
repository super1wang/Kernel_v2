cmake_minimum_required(VERSION 3.25)
project(OCKB4InstalledConsumer LANGUAGES CXX)
find_package(OCK CONFIG REQUIRED COMPONENTS Runtime Adapter::CpuPool)
if(NOT OCK_SDK_VERSION STREQUAL "0.1.0-dev.6" OR NOT OCK_IMPLEMENTATION_STAGE STREQUAL "B5Subset" OR NOT OCK_INSTALLATION_PROFILE MATCHES "^B[45]Subset$")
  message(FATAL_ERROR "B4 installed SDK metadata mismatch")
endif()
get_target_property(cpu_kind OCK::Adapter::CpuPool TYPE)
get_target_property(cpu_links OCK::Adapter::CpuPool INTERFACE_LINK_LIBRARIES)
if(NOT cpu_kind STREQUAL "STATIC_LIBRARY" OR NOT cpu_links STREQUAL "OCK::CoreContracts")
  message(FATAL_ERROR "CpuPool must export only the CoreContracts closure")
endif()
add_executable(installed_b4 main.cpp)
target_link_libraries(installed_b4 PRIVATE OCK::Runtime OCK::Adapter::CpuPool)
target_link_options(installed_b4 PRIVATE /MAP)
