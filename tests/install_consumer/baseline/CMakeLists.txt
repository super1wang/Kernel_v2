cmake_minimum_required(VERSION 3.25)
project(OCKInstalledMetadataConsumer LANGUAGES CXX)
if(REQUIRE_COMPONENT)
  find_package(OCK 0.1.0 CONFIG REQUIRED COMPONENTS ${REQUIRE_COMPONENT})
else()
  find_package(OCK 0.1.0 CONFIG REQUIRED)
endif()
if(NOT OCK_RUNTIME_AVAILABLE)
  message(FATAL_ERROR "SDK Runtime missing")
endif()
get_target_property(stage OCK::Runtime OCK_IMPLEMENTATION_STAGE)
if(NOT stage STREQUAL "B6Subset")
  message(FATAL_ERROR "Missing B6Subset target metadata")
endif()
if(TARGET OCK::ControlClient)
  get_target_property(client_links OCK::ControlClient INTERFACE_LINK_LIBRARIES)
  get_target_property(client_stage OCK::ControlClient OCK_IMPLEMENTATION_STAGE)
  set(expected_client_links "OCK::ControlProtocol;$<LINK_ONLY:bcrypt>")
  if(NOT client_stage STREQUAL "B3Subset" OR NOT "${client_links}" STREQUAL "${expected_client_links}")
    message(FATAL_ERROR "ControlClient export differs from declared client/system dependencies")
  endif()
endif()
add_executable(installed_metadata ../../compile/sdk_metadata.cpp)
target_link_libraries(installed_metadata PRIVATE OCK::Foundation)
