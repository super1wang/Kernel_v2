cmake_minimum_required(VERSION 3.25)
project(OCK VERSION 0.1.0 LANGUAGES CXX)
set(OCK_SDK_VERSION "0.1.0-dev.1")
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
include(cmake/TargetDependencies.cmake)
ock_write_target_graph("${CMAKE_BINARY_DIR}/ock-target-graph.json")
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/OCKConfig.cmake.in "${CMAKE_BINARY_DIR}/OCKConfig.cmake" INSTALL_DESTINATION lib/cmake/OCK)
write_basic_package_version_file("${CMAKE_BINARY_DIR}/OCKConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(EXPORT OCKTargets NAMESPACE OCK:: DESTINATION lib/cmake/OCK)
install(FILES "${CMAKE_BINARY_DIR}/OCKConfig.cmake" "${CMAKE_BINARY_DIR}/OCKConfigVersion.cmake" DESTINATION lib/cmake/OCK)
install(FILES packages/foundation/include/ock/foundation/sdk_version.hpp DESTINATION include/ock/foundation)
install(FILES sdk/sdk_api_manifest.json DESTINATION share/ock)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  add_executable(ock_sdk_metadata_probe tests/compile/sdk_metadata.cpp)
  target_link_libraries(ock_sdk_metadata_probe PRIVATE OCK::Foundation)
  add_test(NAME T24.sdk.version_header COMMAND ock_sdk_metadata_probe)
  add_test(NAME T01.architecture.dependency_dag COMMAND "${Python3_EXECUTABLE}" -X utf8 "${PROJECT_SOURCE_DIR}/tools/architecture/check.py" --graph "${CMAKE_BINARY_DIR}/ock-target-graph.json")
  add_test(NAME T01.architecture.negative_guards COMMAND "${Python3_EXECUTABLE}" -X utf8 -m unittest discover -s "${PROJECT_SOURCE_DIR}/tests/architecture" -v)
  set_tests_properties(T01.architecture.negative_guards PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
  add_test(NAME T24.sdk.install_and_missing_components COMMAND "${Python3_EXECUTABLE}" -X utf8 "${PROJECT_SOURCE_DIR}/tests/install_consumer/verify_baseline.py" --build "${CMAKE_BINARY_DIR}" --config "$<CONFIG>")
  set_tests_properties(T24.sdk.install_and_missing_components PROPERTIES TIMEOUT 180)
  set_tests_properties(T24.sdk.version_header T01.architecture.dependency_dag T01.architecture.negative_guards T24.sdk.install_and_missing_components PROPERTIES LABELS "D0.02")
  option(OCK_BUILD_EXECUTION_MODEL_TESTS "Enable D0.03 reference-model tests" OFF)
  option(OCK_BUILD_G0_TESTS "Enable all implemented G0 contract and evidence tests" OFF)
  if(OCK_BUILD_EXECUTION_MODEL_TESTS OR OCK_BUILD_G0_TESTS)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -X utf8 "${PROJECT_SOURCE_DIR}/tools/testing/model_cases.py" --emit-cmake "${CMAKE_BINARY_DIR}/D0.03Tests.cmake" RESULT_VARIABLE model_discovery_result)
    if(NOT model_discovery_result EQUAL 0)
      message(FATAL_ERROR "D0.03 model discovery failed")
    endif()
    include("${CMAKE_BINARY_DIR}/D0.03Tests.cmake")
  endif()
  option(OCK_BUILD_DEPENDENCY_PROBES "Enable locked development-only dependency probes" OFF)
  if(OCK_BUILD_DEPENDENCY_PROBES)
    include(cmake/Dependencies.cmake)
    include(cmake/DependencyProbes.cmake)
    ock_add_dependency_probes()
  endif()
  if(OCK_BUILD_G0_TESTS)
    add_test(NAME T01.requirements.traceability COMMAND "${Python3_EXECUTABLE}" -X utf8 "${PROJECT_SOURCE_DIR}/tools/requirements/check.py")
    add_test(NAME T01.requirements.negative_guards COMMAND "${Python3_EXECUTABLE}" -X utf8 -m unittest discover -s "${PROJECT_SOURCE_DIR}/tests/requirements" -v)
    set_tests_properties(T01.requirements.traceability T01.requirements.negative_guards PROPERTIES LABELS "D0.01" WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 60)
    add_test(NAME T23.adapter.contract_guards COMMAND "${Python3_EXECUTABLE}" -X utf8 -m unittest discover -s "${PROJECT_SOURCE_DIR}/tests/tools/bootstrap" -v)
    set_tests_properties(T23.adapter.contract_guards PROPERTIES LABELS "D0.06" WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 90)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -X utf8 "${PROJECT_SOURCE_DIR}/tests/plan/run_cases.py" --emit-cmake "${CMAKE_BINARY_DIR}/D0.04Tests.cmake" RESULT_VARIABLE plan_discovery_result)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -X utf8 "${PROJECT_SOURCE_DIR}/tools/testing/g0_cases.py" --emit-cmake "${CMAKE_BINARY_DIR}/G0Tests.cmake" RESULT_VARIABLE g0_discovery_result)
    if(NOT plan_discovery_result EQUAL 0 OR NOT g0_discovery_result EQUAL 0)
      message(FATAL_ERROR "G0 actual test discovery failed")
    endif()
    include("${CMAKE_BINARY_DIR}/D0.04Tests.cmake")
    include("${CMAKE_BINARY_DIR}/G0Tests.cmake")
  endif()
  get_property(_ock_tests DIRECTORY PROPERTY TESTS)
  set_property(TEST ${_ock_tests} APPEND PROPERTY ENVIRONMENT "MSBUILDDISABLENODEREUSE=1")
endif()
