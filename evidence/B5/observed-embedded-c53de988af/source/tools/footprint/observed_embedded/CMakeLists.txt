cmake_minimum_required(VERSION 3.25)
project(OCKObservedEmbedded LANGUAGES CXX)
find_package(OCK CONFIG REQUIRED COMPONENTS Runtime Adapter::CpuPool)
if(NOT OCK_INSTALLATION_PROFILE STREQUAL "Embedded")
  message(FATAL_ERROR "A real Embedded installation is required")
endif()
add_executable(observed_embedded main.cpp)
target_compile_features(observed_embedded PRIVATE cxx_std_20)
target_compile_options(observed_embedded PRIVATE /utf-8 /EHsc /Zc:__cplusplus /permissive-)
target_link_options(observed_embedded PRIVATE /MAP)
set_property(TARGET observed_embedded PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
target_link_libraries(observed_embedded PRIVATE OCK::Runtime OCK::Adapter::CpuPool)
