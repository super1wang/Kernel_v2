cmake_minimum_required(VERSION 3.25)
project(OCKEmbeddedFootprint LANGUAGES CXX)
find_package(OCK CONFIG REQUIRED COMPONENTS Runtime Adapter::CpuPool)
if(NOT OCK_INSTALLATION_PROFILE STREQUAL "Embedded")
  message(FATAL_ERROR "A real Embedded installation is required")
endif()
foreach(kind baseline embedded)
  add_executable(footprint_${kind} main.cpp)
  target_compile_features(footprint_${kind} PRIVATE cxx_std_20)
  target_compile_options(footprint_${kind} PRIVATE /utf-8 /EHsc /Zc:__cplusplus /permissive-)
  target_link_options(footprint_${kind} PRIVATE /MAP)
  set_property(TARGET footprint_${kind} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
endforeach()
target_compile_definitions(footprint_embedded PRIVATE OCK_EMBEDDED_CONSUMER=1)
target_link_libraries(footprint_embedded PRIVATE OCK::Runtime OCK::Adapter::CpuPool)
