cmake_minimum_required(VERSION 3.25)
project(AsanCrtDiagnosis LANGUAGES CXX)
string(REGEX REPLACE "/RTC[1su]" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
foreach(t plain asan)
 add_executable(${t} diagnostic.cpp allocation_probe.cpp)
 target_compile_features(${t} PRIVATE cxx_std_20)
 target_compile_options(${t} PRIVATE /utf-8 /EHsc)
 target_link_options(${t} PRIVATE /INCREMENTAL:NO)
endforeach()
target_compile_options(asan PRIVATE /fsanitize=address)
target_compile_definitions(asan PRIVATE DIAG_ASAN=1)
