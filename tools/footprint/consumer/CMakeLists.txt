cmake_minimum_required(VERSION 3.25)
project(OCKFootprintPair LANGUAGES CXX)
set(OCK_FOOTPRINT_MODE "occupancy" CACHE STRING "Fixed measurement mode")
if(NOT OCK_FOOTPRINT_MODE MATCHES "^(occupancy|allocation)$")
  message(FATAL_ERROR "Invalid footprint mode")
endif()
find_package(OCK 0.1.0 CONFIG REQUIRED COMPONENTS Runtime)
if(NOT OCK_IMPLEMENTATION_STAGE STREQUAL "NativeSubset" OR NOT OCK_RUNTIME_AVAILABLE)
  message(FATAL_ERROR "NativeSubset SDK required")
endif()
get_target_property(_kind OCK::Runtime TYPE)
get_target_property(_links OCK::Runtime INTERFACE_LINK_LIBRARIES)
if(NOT _kind STREQUAL "STATIC_LIBRARY" OR NOT _links STREQUAL "OCK::CoreContracts;$<LINK_ONLY:bcrypt>")
  message(FATAL_ERROR "Unexpected Native Runtime link closure")
endif()
foreach(_variant baseline native)
  add_executable(footprint_${_variant} pair_main.cpp)
  target_compile_features(footprint_${_variant} PRIVATE cxx_std_20)
  target_compile_options(footprint_${_variant} PRIVATE /utf-8 /EHsc /Zc:__cplusplus /permissive-)
  set_property(TARGET footprint_${_variant} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
endforeach()
target_compile_definitions(footprint_native PRIVATE OCK_FOOTPRINT_NATIVE=1)
target_link_libraries(footprint_native PRIVATE OCK::Runtime)
if(OCK_FOOTPRINT_MODE STREQUAL "allocation")
  foreach(_variant baseline native)
    target_compile_definitions(footprint_${_variant} PRIVATE OCK_FOOTPRINT_ALLOCATION=1)
    target_sources(footprint_${_variant} PRIVATE "../../../tests/contract/native/allocation_probe.cpp")
  endforeach()
  add_executable(footprint_native_injected pair_main.cpp "../../../tests/contract/native/allocation_probe.cpp")
  target_link_libraries(footprint_native_injected PRIVATE OCK::Runtime)
  target_compile_features(footprint_native_injected PRIVATE cxx_std_20)
  target_compile_options(footprint_native_injected PRIVATE /utf-8 /EHsc /Zc:__cplusplus /permissive-)
  target_compile_definitions(footprint_native_injected PRIVATE OCK_FOOTPRINT_NATIVE=1 OCK_FOOTPRINT_ALLOCATION=1 OCK_FOOTPRINT_INJECT_ALLOCATION=1)
  set_property(TARGET footprint_native_injected PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
endif()
