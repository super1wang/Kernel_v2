cmake_minimum_required(VERSION 3.25)
project(OCKStatelessService LANGUAGES CXX)
find_package(OCK 0.1.0 CONFIG REQUIRED COMPONENTS Runtime)
if(NOT OCK_SDK_VERSION STREQUAL "0.1.0-dev.3" OR NOT OCK_IMPLEMENTATION_STAGE STREQUAL "B2Subset" OR NOT OCK_RUNTIME_AVAILABLE)
  message(FATAL_ERROR "NativeSubset SDK identity mismatch")
endif()
get_target_property(_kind OCK::Runtime TYPE)
get_target_property(_links OCK::Runtime INTERFACE_LINK_LIBRARIES)
if(NOT _kind STREQUAL "STATIC_LIBRARY" OR NOT _links STREQUAL "OCK::CoreContracts;$<LINK_ONLY:bcrypt>")
  message(FATAL_ERROR "NativeSubset actual static link closure mismatch")
endif()
add_executable(ock_stateless_service main.cpp)
target_link_libraries(ock_stateless_service PRIVATE OCK::Runtime)
